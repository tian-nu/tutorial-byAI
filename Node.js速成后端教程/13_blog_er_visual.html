users

id PK INTEGER
username VARCHAR(50)
email VARCHAR(100)
password_hash VARCHAR(255)
created_at DATETIME

articles

id PK INTEGER
title VARCHAR(200)
content TEXT
author_id FK INTEGER
cover_image VARCHAR(255)
created_at DATETIME
updated_at DATETIME

comments

id PK INTEGER
content TEXT
article_id FK INTEGER
user_id FK INTEGER
created_at DATETIME
PK 主键 (PRIMARY KEY) FK 外键 (FOREIGN KEY) 粗体 NOT NULL
